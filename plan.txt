function focusLocation(orgId, options = {}) {
  const {
    openPopup = true,
    keepCurrentZoom = false
  } = options;

  const marker = markerByOrgId.get(String(orgId));
  if (!marker) return;

  selectedOrgId = orgId;
  shouldAutoFitMap = false;

  const selectedRow = filteredRows.find(r => String(r.org_id) === String(orgId));

  renderLocationList(filteredRows);
  scrollSelectedCardIntoView();

  const latLng = marker.getLatLng();
  const nextZoom = keepCurrentZoom ? map.getZoom() : Math.max(map.getZoom(), FOCUSED_ZOOM);

  // 🔥 OFFSET the map so popup is visible
  const offsetY = 120; // pixels upward shift
  const point = map.project(latLng, nextZoom).subtract([0, offsetY]);
  const newLatLng = map.unproject(point, nextZoom);

  map.setView(newLatLng, nextZoom, { animate: true });

  if (openPopup) marker.openPopup();

  // 🔥 NEW: update route if user location exists
  if (closestSession && closestSession.userLat && selectedRow) {
    closestSession.closestRow = selectedRow;
    renderClosestSessionVisuals();
  }
}
2. Improve Route Sync Behavior
Reason
Ensure only ONE route exists and updates cleanly

function renderClosestSessionVisuals() {
  routeLayer.clearLayers();

  if (!closestSession) return;

  const { userLat, userLng, closestRow } = closestSession;
  if (!closestRow) return;

  const routeColor = getDayColor(closestRow.meeting_day);

  const userMarker = L.marker([userLat, userLng], {
    icon: buildUserMarkerIcon()
  }).bindPopup('<div class="user-popup-label">Your location</div>');

  const line = L.polyline(
    [
      [userLat, userLng],
      [closestRow.lat, closestRow.lng]
    ],
    {
      color: routeColor,
      weight: 4,
      opacity: 0.9,
      dashArray: '8 6',
      lineCap: 'round'
    }
  );

  line.addTo(routeLayer);
  userMarker.addTo(routeLayer);
}
3. Popup Redesign (MAJOR VISUAL UPGRADE)
Reason

Unify with sidebar + improve readability

Replace buildPopupHtml() entirely:
function buildPopupHtml(row) {
  const name = escapeHtml(row.name || 'Unnamed location');
  const address = escapeHtml(row.address || 'Address unavailable');
  const meetingLine = escapeHtml(formatMeetingLine(row));
  const notes = row.notes ? escapeHtml(row.notes) : '';
  const website = row.website ? String(row.website).trim() : '';
  const mapsUrl = buildGoogleMapsUrl(row);

  const dayConfig = getDayConfig(row.meeting_day);
  const color = getDayColor(row.meeting_day);

  const distanceHtml = Number.isFinite(row.distanceKm)
    ? `<div class="popup-distance">${escapeHtml(formatDistanceKm(row.distanceKm))}</div>`
    : '';

  const websiteHtml = website
    ? `<a class="popup-btn" href="${escapeHtml(website)}" target="_blank">Website</a>`
    : '';

  return `
    <div class="popup-card" style="border-left: 5px solid ${color}">
      <div class="popup-title">${name}</div>
      <div class="popup-address">${address}</div>
      <div class="popup-meeting">${meetingLine}</div>
      ${distanceHtml}
      ${notes ? `<div class="popup-notes">${notes}</div>` : ''}

      <div class="popup-actions">
        <a class="popup-btn" href="${escapeHtml(mapsUrl)}" target="_blank">Map</a>
        ${websiteHtml}
      </div>
    </div>
  `;
}
